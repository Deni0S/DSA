## Синтаксис языка C
```c
main //точка входа в систему
<> //путь где находится компилятор
"" //путь относительно папки проекта
#ifndef //если еще не описаны
#define //описать
#endif //конец if
#define true 1 //заменит все true еденицами
void //функция которая ничего не возвращает
\n //новая строка
\t //табуляция
\\ //обратный слешь
\0 //конец строки
%d //int (значение)
%s //строка
%c //char (символы)
%f //float
%lf //long float (double)
%.2f //количество знаков после запятой
%05d //общее количество знаков целого числа (без нуля пробелы)
%% //текст символа %
%p //указатели (&a - адрес / int * pointer (указатель); pointer = &a  / *a - разиминовать, то есть получить доступ к переменной на которую ссылается указатель)
unsigned //беззнаковый
int: 4 bytes /  //2147000000…2147000000
unsigned int: 4 bytes //0...4294000000
short: 2 bytes  //-32768…+32767
double (long float): 8 bytes //матиса экспонента и знак
char: 1 byte //128…+127
unsigned char: 1 byte //0…+255
char sir[25] = "строка"
char* sir = "строка"
puts(sir) //вывод строки
gets() //считать строку из консоли
strcat(sir, name) //кокатенация (склеивание строк)
strlen() //количество символов в строке
isalpha() //если ли символ из алфавита вернет true
isdigit() //проверка на десятичную  цифру
isspace() //проверка на пробел
isupper() //проверка на верхний регистр
islower() //проверка на нижний регистр
toupper() //перевести в верхний режим
tolower() //перевести в нижний режим
atoi() //перевод в численное значение int
atof() //перевод в численное значение float
fopen() //возвращает указатель на адрес в памяти (r - read, w - write, a - append, rb, wb, ab)
fprintf() //запись в файл
fscanf() //прочитать форматированное значение из файла
fclose() //закрыть файл
feof() //возвращает ноль если конец файла не достигнут
? "истина" : "ложь" //тернарный оператор, вернет соответствие условию
break //прекратить выполнение цикла
continue //прекратить текущую итерацию цикла, передаст управление в логическую конструкцию цикла
malloc() //выделение (резервирование) памяти
calloc(кол элементов, размер) //очистить память перед созданием
sizeof() //возвращает размер переменных в байтах
free() //освободить память
realloc() //изменить размер памяти
exit() //немедленное завершение программы и очистка памяти
int * area = (int*) malloc (sizeof(int) * 10) = int area [10] //почти одно и то же
‘\0’ //конец строки
/*  */ //закомментировать
& //и
| //или
! //не
^ //исключающее или
««  »» //сдвиги
4%2  //остаток от деления 4 на 2
++ //инкримент увеличивает на 1
-- //декремент уменьшает на 1
> //больше
< //меньше
>= //больше или равно
<= //меньше или равно
== //проверка на равенство
&& //и (оба истина)
|| //или (хотя бы один истина)
! //не (ложный)
^ //исключающее или (когда различаются)
^= //иксор
break //выход из цикла
continue //пропустить
Exit //выйти из программы
int array1[число элементов] array1[0] = 9; / {0, 2 , 4} //массив
#include <stdlib.h> srand(time(NULL)) //генератор случайных чисел
```
